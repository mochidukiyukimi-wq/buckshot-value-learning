cmake_minimum_required(VERSION 3.20)
project(roulette LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
add_library(roulette_core STATIC
    cpp/src/ammo_belief.cpp cpp/src/chance.cpp cpp/src/rules.cpp cpp/src/state_features.cpp
    cpp/src/sampling.cpp cpp/src/state_key.cpp cpp/src/search_session.cpp
    cpp/src/backup.cpp cpp/src/shortcuts.cpp)
target_include_directories(roulette_core PUBLIC cpp/include)
set_target_properties(roulette_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(roulette_core PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(roulette_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
pybind11_add_module(_native cpp/bindings/module.cpp)
target_link_libraries(_native PRIVATE roulette_core)
install(TARGETS _native LIBRARY DESTINATION roulette RUNTIME DESTINATION roulette)
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests cpp/tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE roulette_core)
    add_test(NAME core_tests COMMAND core_tests)
endif()
